{{- /* Images keep their own aspect ratio; block images become a figure linking to the full-size file (lightbox in post.js). Caption comes from the optional Markdown title ("...") or from non-hash alt text. */ -}} {{- $u := urls.Parse .Destination -}} {{- $src := $u.String -}} {{- $width := 0 -}} {{- $height := 0 -}} {{- if not $u.IsAbs -}} {{- $path := strings.TrimPrefix "./" $u.Path -}} {{- with or (.PageInner.Resources.Get $path) (resources.Get $path) -}} {{- $src = .RelPermalink -}} {{- if in (slice "jpeg" "png" "gif" "webp") .MediaType.SubType -}} {{- $width = .Width -}}{{- $height = .Height -}} {{- end -}} {{- else -}} {{- $file := path.Join "static" $u.Path -}} {{- if and (strings.HasPrefix $u.Path "/") (fileExists $file) (findRE `(?i)\.(jpe?g|png|gif|webp)$` $u.Path 1) -}} {{- with images.Config $file -}}{{- $width = .Width -}}{{- $height = .Height -}}{{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $altText := .Text | strings.TrimSpace -}} {{- $isHashAlt := gt (len (findRE `^[0-9a-fA-F]{16,}$` $altText 1)) 0 -}} {{- $caption := .Title | default (cond (not $isHashAlt) $altText "") -}} {{- $img := printf `` $src ($altText | transform.HTMLEscape) (cond (gt $width 0) (printf ` width="%d" height="%d"` $width $height) "") (cond (ne .Title "") (printf ` title="%s"` (.Title | transform.HTMLEscape)) "") -}} {{- if .IsBlock -}} {{- else -}} {{ $img | safeHTML }} {{- end -}}
